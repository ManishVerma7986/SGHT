import { useState } from "react"
import { usePrivy, useWallets } from "@privy-io/react-auth"

import LoginPage from "./features/auth/LoginPage"
import RegistrationForm from "./features/auth/RegistrationForm"
import MemberDashboard from "./features/dashboard/MemberDashboard"

import { useBackendIdentity } from "./hooks/useBackendIdentity"
import { useRegistration } from "./hooks/useRegistration"

import "./App.css"

function App() {
  const {
    ready,
    authenticated,
    user,
    login,
    logout,
    linkPasskey,
    getAccessToken,
  } = usePrivy()

  const { wallets } = useWallets()

  const [linking, setLinking] = useState(false)
  const [message, setMessage] = useState("")
  const [activeTab, setActiveTab] = useState("Overview")

  const wallet = wallets.find(
    (item) => item.walletClientType === "privy"
  )

  const hasPasskey = user?.linkedAccounts?.some(
    (account) => account.type === "passkey"
  ) || false

  const {
    identity,
    loading: backendLoading,
    error: backendError,
    accessDenied,
    refreshIdentity,
    retry,
  } = useBackendIdentity({
    ready,
    authenticated,
    getAccessToken,
  })

  const {
    loading: registrationLoading,
    error: registrationError,
    submitRegistration,
  } = useRegistration({
    getAccessToken,
    refreshIdentity,
  })

  const handleLinkPasskey = async () => {
    setLinking(true)
    setMessage("")

    try {
      await linkPasskey()
      setMessage("Passkey linked successfully!")
    } catch (error) {
      console.error("Passkey linking failed:", error)
      setMessage(error.message || "Passkey linking failed.")
    } finally {
      setLinking(false)
    }
  }

  const handleRegistration = async (memberData) => {
    try {
      console.log("APP RECEIVED REGISTRATION:", memberData)
      await submitRegistration(memberData)
    } catch {
      // The registration hook maintains and exposes the error state.
    }
  }

  if (!ready) {
    return (
      <main className="loading-screen">
        <div className="loading-orb" />
        <p>Preparing your SHGT experience...</p>
      </main>
    )
  }

  if (!authenticated) {
    return <LoginPage onLogin={login} />
  }

  if (backendLoading) {
    return (
      <main className="loading-screen">
        <div className="loading-orb" />
        <p>Verifying your SHGT identity...</p>
      </main>
    )
  }

  if (accessDenied) {
    return (
      <main className="loading-screen">
        <h2>Access restricted</h2>
        <p>
          Your authenticated identity does not currently have
          permission to access this SHGT account.
        </p>
        <button className="primary-button" onClick={logout}>
          Sign out
        </button>
      </main>
    )
  }

  if (backendError || !identity) {
    return (
      <main className="loading-screen">
        <h2>Unable to verify your identity</h2>

        <p role="alert">
          {backendError || "No identity information was returned by SHGT."}
        </p>

        <button className="primary-button" onClick={retry}>
          Retry verification
        </button>

        <button className="logout-button" onClick={logout}>
          Sign out
        </button>
      </main>
    )
  }
  console.log("SHGT FRONTEND STATE:", {
    authenticated,
    backendLoading,
    backendError,
    identity,
    accessDenied,
  })

  if (!identity.registered) {
    return (
      <RegistrationForm
        initialEmail={user?.email?.address || ""}
        loading={registrationLoading}
        error={registrationError}
        onSubmit={handleRegistration}
        onLogout={logout}
      />
    )
  }

  return (
    <MemberDashboard
      user={user}
      wallet={wallet}
      hasPasskey={hasPasskey}
      linking={linking}
      message={message}
      onLinkPasskey={handleLinkPasskey}
      onLogout={logout}
      activeTab={activeTab}
      setActiveTab={setActiveTab}
      backendIdentity={identity}
      backendLoading={backendLoading}
      backendError={backendError}
    />
  )
}

export default App

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL

if (!API_BASE_URL) {
  throw new Error("VITE_API_BASE_URL is not configured.")
}

export async function apiRequest(path, options = {}) {
  const {
    token,
    headers = {},
    ...requestOptions
  } = options

  const response = await fetch(
    `${API_BASE_URL}${path}`,
    {
      ...requestOptions,
      headers: {
        ...(requestOptions.body ? { "Content-Type": "application/json" } : {}),
        ...headers,
        ...(token ? { Authorization: `Bearer ${token}` } : {}),
      },
    }
  )

  const data = await response.json().catch(() => ({}))

  if (!response.ok) {
    const error = new Error(
      data.message || `API request failed (${response.status}).`
    )

    error.status = response.status
    error.data = data

    throw error
  }

  return data
}

    token,
  })
}

export function registerMember(token, memberData) {
  return apiRequest("/api/v1/auth/register", {
    method: "POST",
    token,
    body: JSON.stringify(memberData),
  })
}

function RegistrationForm({
  initialEmail = "",
  loading,
  error,
  onSubmit,
  onLogout,
}) {
  const [fullName, setFullName] = useState("")
  const [phoneNumber, setPhoneNumber] = useState("")
  const [email, setEmail] = useState(initialEmail)

  const handleSubmit = async (event) => {
    event.preventDefault()
    console.log("REGISTRATION FORM SUBMITTED")

    await onSubmit({
      fullName: fullName.trim(),
      phoneNumber: phoneNumber.trim(),
      email: email.trim(),
    })
  }

  return (
    <main className="registration-shell">
      <div className="registration-card">
        <div className="brand">
          <span className="brand-symbol">S</span>
          <span>
            SHGT<span className="brand-dot">.</span>
          </span>
        </div>

        <div className="card-eyebrow">MEMBER REGISTRATION</div>

        <h1>Complete your SHGT profile</h1>

        <p>
          Your Privy identity has been authenticated. Add your basic
          details to create your SHGT member account.
        </p>

        <form onSubmit={handleSubmit}>
          <label htmlFor="fullName">Full name *</label>
          <input
            id="fullName"
            type="text"
            value={fullName}
            onChange={(event) => setFullName(event.target.value)}
            minLength={2}
            maxLength={150}
            autoComplete="name"
            required
            disabled={loading}
          />

          <label htmlFor="phoneNumber">Phone number *</label>

async function initializePool() {
    if (pool) {
        return pool;
    }

    const walletLocation = path.resolve(
        __dirname,
        "../..",
        process.env.DB_WALLET_LOCATION
    );

    try {
        pool = await oracledb.createPool({
            user: process.env.DB_USER,
            password: process.env.DB_PASSWORD,
            connectString: process.env.DB_CONNECT_STRING,

            configDir: walletLocation,
            walletLocation: walletLocation,
            walletPassword: process.env.DB_WALLET_PASSWORD,

            poolMin: 1,
            poolMax: 5,
            poolIncrement: 1,
            poolTimeout: 60
        });

        console.log("Oracle connection pool initialized.");

        return pool;
    } catch (error) {
        console.error(
            "Oracle connection pool initialization failed:",
            error.message
        );

        throw error;
    }
}

async function getConnection() {
    if (!pool) {
        throw new Error("Oracle connection pool is not initialized.");
    }

    return pool.getConnection();
}

    if (!pool) {
        await initializePool();
    }

    return getConnection();
}

const { PrivyClient, InvalidAuthTokenError } = require("@privy-io/node");

const appId = process.env.PRIVY_APP_ID;
const appSecret = process.env.PRIVY_APP_SECRET;

if (!appId || !appSecret) {
    throw new Error("Missing PRIVY_APP_ID or PRIVY_APP_SECRET.");
}

const privy = new PrivyClient({
    appId,
    appSecret
});

async function requirePrivyAuth(req, res, next) {
    const authorization = req.headers.authorization;

    if (!authorization || !authorization.startsWith("Bearer ")) {
        return res.status(401).json({
            success: false,
            message: "Authentication token required."
        });
    }

    const accessToken = authorization.slice(7).trim();

    if (!accessToken) {
        return res.status(401).json({
            success: false,
            message: "Authentication token is empty."
        });
    }

    try {
        const claims = await privy
            .utils()
            .auth()
            .verifyAccessToken(accessToken);

        if (!claims.user_id) {
            return res.status(401).json({
                success: false,
                message: "Invalid authentication identity."
            });
        }

        // This ID comes from the verified token, not the request body.
        req.privyUserId = claims.user_id;

        next();
    } catch (error) {
        if (
            error instanceof InvalidAuthTokenError ||
            error.name === "InvalidAuthTokenError"
        ) {
            return res.status(401).json({
                success: false,
                message: "Invalid or expired authentication token."
            });
        }

        // A verifier/network failure is not automatically an invalid token.
        console.error("Privy verification service error:", error.message);

        return res.status(503).json({
            success: false,
            message: "Authentication service temporarily unavailable."
        });
    }
}

module.exports = requirePrivyAuth;

const express = require("express");
const oracledb = require("oracledb");

const router = express.Router();

const requirePrivyAuth = require("../middleware/requirePrivyAuth");
const { getConnection } = require("../config/database");

// Find an SHGT account using the verified Privy identity.
async function findExistingMember(connection, privyUserId) {
    const result = await connection.execute(
        `SELECT USER_ID, ACCOUNT_STATUS
         FROM SHGT_APP.USERS
         WHERE PRIVY_USER_ID = :privyUserId`,
        { privyUserId }
    );

    if (result.rows.length === 0) {
        return null;
    }

    const [userId, accountStatus] = result.rows[0];

    return { userId, accountStatus };
}

// Return the actual state of an existing account.
function respondWithExistingMember(res, member) {
    if (member.accountStatus !== "ACTIVE") {
        return res.status(403).json({
            success: false,
            authenticated: true,
            registered: true,
            userId: member.userId,
            accountStatus: member.accountStatus,
            message: "This SHGT account is not active."
        });
    }

    return res.status(200).json({
        success: true,
        authenticated: true,
        registered: true,
        userId: member.userId,
        accountStatus: member.accountStatus,
        message: "This Privy identity is already registered with SHGT."
    });
}


// GET /api/v1/auth/me
// Checks whether the authenticated Privy identity is registered.
router.get("/me", requirePrivyAuth, async (req, res) => {
    let connection;

    try {
        connection = await getConnection();

        const member = await findExistingMember(
            connection,
            req.privyUserId
        );

        if (!member) {
            return res.json({
                success: true,
                authenticated: true,
                registered: false,
                message: "Privy identity verified. SHGT registration is pending."
            });
        }

        return respondWithExistingMember(res, member);

    } catch (error) {
        console.error(
            "Authenticated identity lookup failed:",
            error.message
        );

        return res.status(500).json({
            success: false,
            message: "Unable to retrieve member identity."
        });

    } finally {
        if (connection) {
            await connection.close();
        }
    }
});


// POST /api/v1/auth/register
// Creates a user, member profile and pending KYC record atomically.
router.post("/register", requirePrivyAuth, async (req, res) => {
    let connection;
    let transactionStarted = false;

    try {
        const {
            fullName,
            phoneNumber,
            email
        } = req.body || {};

        // Validate full name.
        if (
            typeof fullName !== "string" ||
            fullName.trim().length < 2 ||
            fullName.trim().length > 150 ||
            /[\u0000-\u001F\u007F]/.test(fullName)
        ) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid full name (2-150 characters)."
            });
        }

        const normalizedName = fullName.trim();

        // Validate and normalize optional phone number.
        let normalizedPhone = null;

        if (phoneNumber !== undefined && phoneNumber !== null) {
            if (typeof phoneNumber !== "string") {
                return res.status(400).json({
                    success: false,
                    message: "Invalid phone number."
                });
            }

            const cleanedPhone = phoneNumber
                .trim()
                .replace(/[\s()-]/g, "");

            if (cleanedPhone.length > 0) {
                if (!/^\+?[1-9]\d{7,14}$/.test(cleanedPhone)) {
                    return res.status(400).json({
                        success: false,
                        message: "Enter a valid international phone number."
                    });
                }

                normalizedPhone = cleanedPhone;
            }
        }

        // Validate and normalize optional email.
        let normalizedEmail = null;

        if (email !== undefined && email !== null) {
            if (typeof email !== "string") {
                return res.status(400).json({
                    success: false,
                    message: "Invalid email address."
                });
            }

            const cleanedEmail = email.trim().toLowerCase();

            if (cleanedEmail.length > 0) {
                if (
                    cleanedEmail.length > 254 ||
                    !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(cleanedEmail)
                ) {
                    return res.status(400).json({
                        success: false,
                        message: "Enter a valid email address."
                    });
                }

                normalizedEmail = cleanedEmail;
            }
        }

        // Identity comes exclusively from verified Privy middleware.
        const privyUserId = req.privyUserId;

        if (
            typeof privyUserId !== "string" ||
            privyUserId.length === 0 ||
            privyUserId.length > 100
        ) {
            return res.status(401).json({
                success: false,
                message: "Invalid authenticated identity."
            });
        }

        connection = await getConnection();

        // Make ordinary registration retries idempotent.
        const existingMember = await findExistingMember(
            connection,
            privyUserId
        );

        if (existingMember) {
            return respondWithExistingMember(res, existingMember);
        }

        // Execute the restricted Oracle registration package.
        // Direct table INSERT privileges are not required.
        transactionStarted = true;

        const result = await connection.execute(
            `BEGIN
                SHGT_APP.AUTH_REGISTRATION_PKG.REGISTER_MEMBER(
                    P_PRIVY_USER_ID => :privyUserId,
                    P_FULL_NAME     => :fullName,
                    P_PHONE_NUMBER  => :phoneNumber,
                    P_EMAIL         => :email,
                    P_USER_ID       => :userId
                );
             END;`,
            {
                privyUserId: {
                    val: privyUserId,
                    type: oracledb.STRING,
                    maxSize: 100
                },
                fullName: {
                    val: normalizedName,
                    type: oracledb.STRING,
                    maxSize: 150
                },
                phoneNumber: {
                    val: normalizedPhone,
                    type: oracledb.STRING,
                    maxSize: 20
                },
                email: {
                    val: normalizedEmail,
                    type: oracledb.STRING,
                    maxSize: 254
                },
                userId: {
                    dir: oracledb.BIND_OUT,
                    type: oracledb.NUMBER
                }
            },
            { autoCommit: false }
        );

        const userId = result.outBinds.userId;

        // Commit only after the entire package succeeds.
        await connection.commit();
        transactionStarted = false;

        return res.status(201).json({
            success: true,
            registered: true,
            userId,
            accountStatus: "ACTIVE",
            verificationStatus: "PENDING",
            message: "SHGT registration completed. Identity verification is pending."
        });

    } catch (error) {
        // Roll back before attempting duplicate recovery.
        let rollbackSucceeded = !transactionStarted;

        if (connection && transactionStarted) {
            try {
                await connection.rollback();
                transactionStarted = false;
                rollbackSucceeded = true;

            } catch (rollbackError) {
                console.error(
                    "Registration rollback failed:",
                    rollbackError.message
                );
            }
        }

        // A unique violation may mean another request registered
        // this same Privy identity while our request was executing.
        if (error.errorNum === 1 && rollbackSucceeded && connection) {
            try {
                const existingMember = await findExistingMember(
                    connection,
                    req.privyUserId
                );

                if (existingMember) {
                    return respondWithExistingMember(
                        res,
                        existingMember
                    );
                }

                // The unique violation was not caused by an existing
                // Privy identity. Do not misreport it as successful.
                return res.status(409).json({
                    success: false,
                    registered: false,
                    message: "Registration conflicts with an existing database record."
                });

            } catch (lookupError) {
                console.error(
                    "Duplicate registration recovery failed:",
                    lookupError.message
                );

                return res.status(500).json({
                    success: false,
                    message: "Unable to verify registration status."
                });
            }
        }

        console.error(
            "SHGT registration failed:",
            error.message
        );

        return res.status(500).json({
            success: false,
            message: "Unable to complete SHGT registration. Please verify your account status before retrying."
        });

    } finally {
        if (connection) {
            try {
                await connection.close();
            } catch (closeError) {
                console.error(
                    "Registration connection cleanup failed:",
                    closeError.message
                );
            }
        }
    }
});
